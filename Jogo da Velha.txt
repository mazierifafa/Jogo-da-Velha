public class jogodavelha {
    
    public static void main(String args[]){
        
        Scanner ler = new Scanner(System.in);
        Random sorteio = new Random();
        
        int[][] tabuleiro = new int[4][4];
        int jogador = 1, vencedor = 0, jogadas = 0;
        int linha, coluna;
        
        while(vencedor == 0 && jogadas < 9){
            
            linha = 0;
            coluna = 0;
            
            System.out.println("Tabuleiro Atual");
            for(int i=1; i<4; i++){
                for(int j=1; j<4; j++){
                    System.out.print(Simbolo(tabuleiro[i][j]) + " ");
                }
                System.out.println();
            }
            
            if(jogador == 1){
                System.out.println("Sua vez!\n");
                System.out.println("Escolha uma linha (1 a 3) e uma coluna (1 a 3):");
                linha = ler.nextInt();
                coluna = ler.nextInt();
                
                if(linha < 1 || linha > 3 || coluna < 1 || coluna > 3){
                    System.out.println("Posicao fora do tabuleiro!");
                    linha = 0;
                    coluna = 0;
                }
                
                else if(tabuleiro[linha][coluna] != 0){
                    System.out.println("Essa posicao ja foi marcada!");
                    linha = 0;
                    coluna = 0;
                }
            }
            
            else{
                System.out.println("Vez do computador...");
                
                while(linha == 0 || coluna == 0){
                    int linhaBot = sorteio.nextInt(3) + 1;
                    int colunaBot = sorteio.nextInt(3) + 1;
                    
                    if(tabuleiro[linhaBot][colunaBot] == 0){
                        linha = linhaBot;
                        coluna = colunaBot;
                    }
                }
                
                System.out.println("O computador jogou na linha " + linha + " e coluna " + coluna);
            }
            
            if(linha != 0){
                tabuleiro[linha][coluna] = jogador;
                jogadas++;
                
                for(int i=1; i<4; i++){
                    if(tabuleiro[i][1] == jogador && tabuleiro[i][2] == jogador && tabuleiro[i][3] == jogador){
                        vencedor = jogador;
                    }
                }
                
                for(int j=1; j<4; j++){
                    if(tabuleiro[1][j] == jogador && tabuleiro[2][j] == jogador && tabuleiro[3][j] == jogador){
                        vencedor = jogador;
                    }
                }
                
                if(tabuleiro[1][1] == jogador && tabuleiro[2][2] == jogador && tabuleiro[3][3] == jogador){
                    vencedor = jogador;
                }
                
                if(tabuleiro[1][3] == jogador && tabuleiro[2][2] == jogador && tabuleiro[3][1] == jogador){
                    vencedor = jogador;
                }
                
                if(vencedor == 0){
                    if(jogador == 1){
                        jogador = 2;
                    }
                    else{
                        jogador = 1;
                    }
                }
            }
            
        }
        
        System.out.println("Tabuleiro Final");
        for(int i=1; i<4; i++){
            for(int j=1; j<4; j++){
                System.out.print(tabuleiro[i][j] + " ");
            }
            System.out.println();
        }
        
        if(vencedor == 1){
            System.out.println("Você venceu o jogo!");
        }
        else if(vencedor == 2){
            System.out.println("O computador venceu o jogo!");
        }
        else{
            System.out.println("O jogo empatou!");
        }
        
    }
    
    public static String Simbolo(int vPos){
        if (vPos == 1) return "X";
        if (vPos == 2) return "O";
        return "-";
    }
    
}
